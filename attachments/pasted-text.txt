    window.getValPeka = function(obj, targetKeys) {
        if (!obj || typeof obj !== 'object') return '';
        const normalizedMap = {};
        for (let key in obj) {
            if (Object.prototype.hasOwnProperty.call(obj, key)) {
                let cleanKey = key.toLowerCase().replace(/[^a-z0-9]/g, '');
                normalizedMap[cleanKey] = obj[key];
            }
        }
        for (let target of targetKeys) {
            let cleanTarget = target.toLowerCase().replace(/[^a-z0-9]/g, '');
            if (normalizedMap[cleanTarget] !== undefined && normalizedMap[cleanTarget] !== null) {
                return String(normalizedMap[cleanTarget]).trim();
            }
        }
        return '';
    };

    // --- HELPER STORAGE INDEXEDDB (KAPASITAS BESAR & CEPAT) ---
    const IDBHelper = {
        dbName: 'TaklimdaDB',
        storeName: 'appCache',
        db: null,
        async init() {
            if (this.db) return this.db;
            return new Promise((resolve, reject) => {
                let request = indexedDB.open(this.dbName, 1);
                request.onupgradeneeded = (e) => {
                    let db = e.target.result;
                    if (!db.objectStoreNames.contains(this.storeName)) {
                        db.createObjectStore(this.storeName);
                    }
                };
                request.onsuccess = (e) => {
                    this.db = e.target.result;
                    resolve(this.db);
                };
                request.onerror = (e) => resolve(null);
            });
        },
        async set(key, val) {
            try {
                let db = await this.init();
                if (!db) { localStorage.setItem(key, JSON.stringify(val)); return; }
                let tx = db.transaction(this.storeName, 'readwrite');
                tx.objectStore(this.storeName).put(val, key);
            } catch(e) {
                try { localStorage.setItem(key, JSON.stringify(val)); } catch(ex){}
            }
        },
        async get(key) {
            try {
                let db = await this.init();
                if (!db) {
                    let ls = localStorage.getItem(key);
                    return ls ? JSON.parse(ls) : null;
                }
                return new Promise((resolve) => {
                    let tx = db.transaction(this.storeName, 'readonly');
                    let req = tx.objectStore(this.storeName).get(key);
                    req.onsuccess = () => resolve(req.result || null);
                    req.onerror = () => resolve(null);
                });
            } catch(e) {
                let ls = localStorage.getItem(key);
                return ls ? JSON.parse(ls) : null;
            }
        }
    };
    </script>

    function decryptPayload(base64Str, key) {
      const encryptedStr = atob(base64Str);
      let decryptedUri = "";
      for (let i = 0; i < encryptedStr.length; i++) {
        decryptedUri += String.fromCharCode(encryptedStr.charCodeAt(i) ^ key.charCodeAt(i % key.length));
      }
      return decodeURIComponent(decryptedUri);
    }

    function parseDates(val) {
        if (!val && val !== 0) return [];
        let str = Array.isArray(val) ? val.join(',') : String(val);
        let parts = str.split(/[\s,;/\\-]+/).filter(Boolean);
        let result = [];

        parts.forEach(part => {
            let digits = part.replace(/\D/g, '');
            if (!digits) return;

            if (digits.length > 2) {
                for (let i = 0; i < digits.length; i += 2) {
                    let chunk = digits.substring(i, i + 2);
                    let num = parseInt(chunk, 10);
                    if (num >= 1 && num <= 31) result.push(num);
                }
            } else {
                let num = parseInt(digits, 10);
                if (num >= 1 && num <= 31) result.push(num);
            }
        });

        return Array.from(new Set(result));
    }

    async function saveLocalCache() {
        if (currentUser && masterKoreksi.length > 0) {
            await IDBHelper.set(`taklimda_cache_${currentUser}`, masterKoreksi);
        }
    }

    async function loadLocalCacheFirst() {
        if (!currentUser) return;
        let cached = await IDBHelper.get(`taklimda_cache_${currentUser}`);
        if (cached) {
            try {
                masterKoreksi = typeof cached === 'string' ? JSON.parse(cached) : cached;
                updateDashboardStats();
                renderKoreksiList();
                updatePendingUI();
            } catch(e) { console.error("Error reading cache", e); }
        }
    }

    async function processPendingQueue() {
        if (isSyncing || !navigator.onLine) return;
        let queue = getPendingQueue();
        if (queue.length === 0) return;

        isSyncing = true;
        showDynamicIsland('sync', '🔄 Memulai Sinkronisasi...');

        let remaining = [...queue];
        let successCount = 0;

        for (let item of queue) {
            try {
                let res = await fetch(APPS_SCRIPT_URL, {
                    method: 'POST',
                    body: JSON.stringify({
                        action: 'updateKoreksi',
                        data: item
                    })
                });
                
                let jsonRes = await res.json();

                if (res.ok && jsonRes.status === 'success') {
                    remaining = remaining.filter(q => q._rowIndex !== item._rowIndex);
                    
                    let mItem = masterKoreksi.find(x => x._rowIndex === item._rowIndex);
                    if (mItem) mItem._unsynced = false;
                    
                    if (jsonRes.archived) {
                        showToast("ℹ️ Beberapa data disimpan ke Arsip Database karena pekan telah berganti.");
                    }
                    
                    successCount++;
                } else if (jsonRes.status === 'error') {
                    remaining = remaining.filter(q => q._rowIndex !== item._rowIndex);
                    showToast("⚠️ " + (jsonRes.message || "Gagal sinkronisasi data"));
                }
            } catch(e) {
                console.error("Gagal sync item:", item, e);
                break;
            }
        }

        savePendingQueue(remaining);
        await saveLocalCache();
        isSyncing = false;

        if (successCount > 0) {
            showDynamicIsland('online', `✅ Sync ${successCount} Data Berhasil`);
            showToast(`✅ ${successCount} Data offline berhasil tersinkronisasi ke pusat!`);
            renderKoreksiList();
        }
    }

    async function manualSync() {
        if (!navigator.onLine) {
            showDynamicIsland('offline', '⚠️ Sinyal Terputus');
            showToast("⚠️ Tidak ada jaringan internet!");
            return;
        }
        showDynamicIsland('sync', '🔄 Sinkronisasi Data...');
        showToast("Memulai sinkronisasi...");
        await processPendingQueue();
        await loadAllInitData(true);
        await fetchKoreksiDataRealtime();
    }

    async function fetchKoreksiDataRealtime() {
        if (!navigator.onLine) {
            showToast("Offline: Menampilkan data tersimpan di HP");
            return;
        }

        let container = document.getElementById('santri-list-container');

        try {
            let initData = await loadAllInitData();

            if (initData && initData.config && initData.config.filterAbsen) {
                currentFilterMode = initData.config.filterAbsen.toLowerCase();
            }

            let bannerEl = document.getElementById('mode-banner-el');
            if (bannerEl) {
                if (currentFilterMode && currentFilterMode !== 'all') {
                    bannerEl.className = `mode-banner ${currentFilterMode}`;
                    bannerEl.innerText = `📢 Mode Koreksi Aktif: FOKUS DATA ${currentFilterMode.toUpperCase()}`;
                    bannerEl.style.display = 'block';
                } else {
                    bannerEl.style.display = 'none';
                }
            }

            if (initData && initData.broadcast) {
                applyBroadcast(initData.broadcast);
            }

            let allRekap = (initData && initData.rekap) ? initData.rekap : [];
            let rawData = allRekap.filter(r => r._source === 'koreksi' || !r._source);

            if (!Array.isArray(rawData) || rawData.length === 0) {
                masterKoreksi = [];
                await saveLocalCache();
                updateDashboardStats();
                container.innerHTML = `<div style="text-align:center; padding:30px 15px; color:var(--text-muted); font-size:12px; background:white; border-radius:18px; border:1px solid var(--border);">Data pekan ini masih kosong / sudah diarsip.</div>`;
                return;
            }

            let filtered = rawData.filter(r => {
                let dom = String(window.getValPeka(r, ['domisili', 'daerah', 'kamar', 'wilayah']) || '').toUpperCase().trim();
                let curr = String(currentUser || '').toUpperCase().trim();
                if (!curr) return true;
                
                if (!dom.startsWith(curr)) return false;
                let nextChar = dom.charAt(curr.length);
                return !nextChar || !/^[A-Z]/.test(nextChar);
            });

            let pendingQueue = getPendingQueue();

            let mappedList = filtered.map((r, i) => {
                let actualRowIndex = (r._rowIndex !== undefined && r._rowIndex !== null && r._rowIndex !== '') ? r._rowIndex : (r.row_index || r.rowIndex || (i + 2));
                let pendingItem = pendingQueue.find(p => p._rowIndex === actualRowIndex);

                let tAlpha = pendingItem ? parseDates(pendingItem.tglAlpha) : parseDates(window.getValPeka(r, ['tgl_alpha', 'tglalpha', 'tanggal_alpha', 'alpha', 'tanggalalpa']));
                let tIzin  = pendingItem ? parseDates(pendingItem.tglIzin)  : parseDates(window.getValPeka(r, ['tgl_izin', 'tglizin', 'tanggal_izin', 'izin', 'tanggalizin']));
                let tSakit = pendingItem ? parseDates(pendingItem.tglSakit) : parseDates(window.getValPeka(r, ['tgl_sakit', 'tglsakit', 'tanggal_sakit', 'sakit', 'tanggalsakit']));
                let statusK = pendingItem ? pendingItem.status : (window.getValPeka(r, ['status_koreksi', 'status', 'koreksi']) || 'Belum Dikoreksi');
                let ket = pendingItem ? pendingItem.keterangan : (window.getValPeka(r, ['keterangan_taklimda', 'keterangan', 'ket']) || '-');

                let isVerified = (statusK.toLowerCase().includes('sesuai') || statusK.toLowerCase().includes('benar') || statusK.toLowerCase().includes('sudah')) && !statusK.toLowerCase().includes('belum');

                return {
                    ...r,
                    _rowIndex: actualRowIndex,
                    _idPps: window.getValPeka(r, ['id_pps', 'idpps', 'id', 'nis']),
                    _nama: window.getValPeka(r, ['nama_santri', 'nama', 'namasantri', 'name']),
                    _domisili: String(window.getValPeka(r, ['domisili', 'daerah', 'kamar'])).toUpperCase(),
                    _jalsah: window.getValPeka(r, ['jalsah', 'kelompok', 'kelas']) || '-',
                    _foto: window.getValPeka(r, ['foto', 'fotourl', 'urlfoto', 'linkfoto']),
                    _tglAlphaArr: tAlpha,
                    _tglIzinArr: tIzin,
                    _tglSakitArr: tSakit,
                    _keterangan: ket,
                    _statusKoreksi: statusK,
                    _isDone: isVerified,
                    _unsynced: !!pendingItem
                };
            });

            if (currentFilterMode === 'alpha') {
                masterKoreksi = mappedList.filter(x => x._tglAlphaArr.length > 0 || x._isDone);
            } else if (currentFilterMode === 'izin') {
                masterKoreksi = mappedList.filter(x => x._tglIzinArr.length > 0 || x._isDone);
            } else if (currentFilterMode === 'sakit') {
                masterKoreksi = mappedList.filter(x => x._tglSakitArr.length > 0 || x._isDone);
            } else {
                masterKoreksi = mappedList;
            }

            await saveLocalCache();
            updateDashboardStats();
            renderKoreksiList();
            updatePendingUI();
            showToast(`Terhubung: ${masterKoreksi.length} Santri Pekan Ini (${currentFilterMode.toUpperCase()})`);
        } catch(e) {
            console.error(e);
            showToast("Menggunakan data lokal HP (Koneksi terganggu)");
        }
    }

    function updateDashboardStats() {

        let selesai = masterKoreksi.filter(x => x._isDone).length;

        document.getElementById('stat-progress-text').innerText = `${selesai} / ${total} Santri Selesai`;
        document.getElementById('stat-percent').innerText = `${percent}%`;
        document.getElementById('progress-bar-fill').style.width = `${percent}%`;

        let countBelumEl = document.getElementById('count-belum');
        if (countBelumEl) countBelumEl.innerText = belum;

        let countSudahEl = document.getElementById('count-sudah');
        if (countSudahEl) countSudahEl.innerText = selesai;
    }

    function renderKoreksiList() {
        let container = document.getElementById('santri-list-container');
        let searchVal = document.getElementById('search-koreksi').value.toUpperCase().trim();
        container.innerHTML = "";

        let listToShow = masterKoreksi.filter(item => {
            let matchFilter = activeFilter === 'belum' ? !item._isDone : item._isDone;
            let matchSearch = true;
            if (searchVal) {
                let haystack = `${item._nama} ${item._idPps} ${item._domisili} ${item.tingkat || ''} ${item.kategori || ''}`.toUpperCase();
                matchSearch = haystack.includes(searchVal);
            }
            return matchFilter && matchSearch;
        });

        if (selectedRoom !== null) {
            let santriDiKamar = listToShow.filter(item => {
                let room = (item._domisili && item._domisili.trim()) ? item._domisili.trim() : 'LAIN-LAIN';
                return room === selectedRoom;
            });

            let headerDiv = document.createElement('div');
            headerDiv.className = 'room-detail-header fade-in';
            headerDiv.innerHTML = `
                <button class="btn-back" onclick="tutupKamar()">← Kembali ke Daftar Kamar</button>
                <div style="text-align:right;">
                    <div style="font-size:13px; font-weight:800; color:var(--text-main);">Kamar ${selectedRoom}</div>
                    <div style="font-size:10px; color:var(--text-muted); font-weight:700;">${santriDiKamar.length} Santri</div>
                </div>
            `;
            container.appendChild(headerDiv);

            if (santriDiKamar.length === 0) {
                let emptyDiv = document.createElement('div');
                emptyDiv.className = 'fade-in';
                emptyDiv.style.cssText = 'text-align:center; padding:30px 15px; color:var(--text-muted); font-size:12px; background:white; border-radius:18px; border:1px solid var(--border);';
                emptyDiv.innerHTML = `
                    <div style="font-size:32px; margin-bottom:8px;">🎉</div>
                    <div style="font-weight:800; font-size:13px; color:var(--text-main); margin-bottom:4px;">Kamar Ini Sudah Selesai!</div>

                let badgeHtml = `<span class="badge-status badge-belum">Belum</span>`;
                
                if (st.includes('sesuai') || st.includes('benar')) {
                    badgeHtml = `<span class="badge-status badge-benar">✔ Sesuai</span>`;
                } else if ((st.includes('sudah') || st.includes('dikoreksi')) && !st.includes('belum')) {
                    badgeHtml = `<span class="badge-status badge-koreksi">✏ Dikoreksi</span>`;
                }

                let unsyncedBadge = item._unsynced ? `<span class="badge-unsynced">⏳ Belum Sync</span>` : '';

                let card = document.createElement('div');
                card.className = 'santri-card';

                card.innerHTML = `
                    <div class="card-top">
                        <div>
                            <div class="santri-nama">${item._nama}</div>
                            <div class="santri-meta">ID: <b>${item._idPps}</b> • ${item._domisili}</div>
                        </div>

                    </div>
                `;
                listDiv.appendChild(card);
            });

        }

        let grouped = {};
        listToShow.forEach(item => {
            let room = (item._domisili && item._domisili.trim()) ? item._domisili.trim() : 'LAIN-LAIN';

        });

        container.appendChild(gridDiv);
    }

    async function markAsSesuaiByRowIndex(rowIndex) {
        const item = masterKoreksi.find(x => x._rowIndex === rowIndex);
        if (!item) return;

        let hasAlpha = item._tglAlphaArr && item._tglAlphaArr.length > 0;
        let currentKet = (item._keterangan || '').trim();

            bukaModalKoreksiByRowIndex(rowIndex);
            setTimeout(() => {
                let inputKet = document.getElementById('edit-ket');
                if (inputKet) {
                    inputKet.style.borderColor = '#e11d48';
                    inputKet.focus();
                }
            }, 300);
            return;
        }

        item._statusKoreksi = 'Sesuai (Benar)';
        item._isDone = true;
        item._unsynced = true;

        await saveLocalCache();
        updateDashboardStats();
        renderKoreksiList();

        let dataPayload = {
            _rowIndex: item._rowIndex,
            _idPps: item._idPps,
            tglSakit: item._tglSakitArr.join(', '),
            tglIzin: item._tglIzinArr.join(', '),
            tglAlpha: item._tglAlphaArr.join(', '),
            keterangan: item._keterangan !== '-' ? item._keterangan : '',
            status: 'Sesuai (Benar)'
        };

        sendOrQueueData(dataPayload);
    }

    async function sendOrQueueData(dataPayload) {
        if (!navigator.onLine) {
            queueForSync(dataPayload);
            showDynamicIsland('offline', '💾 Data Tersimpan di HP');
            showToast("💾 Tersimpan di HP. Akan dikirim otomatis saat online.");
            return;
        }

        showDynamicIsland('sync', '📤 Mengirim ke Pusat...');
        try {
            let res = await fetch(APPS_SCRIPT_URL, {
                method: 'POST',
                body: JSON.stringify({
                    action: 'updateKoreksi',
                    data: dataPayload
                })
            });
            
            let jsonRes = await res.json();
            
            if (res.ok && jsonRes.status === 'success') {
                showDynamicIsland('online', '✔ Data Sukses Terkirim');
                let mItem = masterKoreksi.find(x => x._rowIndex === dataPayload._rowIndex);
                if (mItem) mItem._unsynced = false;
                
                if (jsonRes.archived) {
                    showToast("ℹ️ Tersimpan ke Arsip Database (Pekan telah berganti).");
                }
                
                await saveLocalCache();
                renderKoreksiList();
            } else if (jsonRes.status === 'error') {
                showToast("⚠️ " + (jsonRes.message || "Server menolak pembaruan data."));
                let queue = getPendingQueue();
                queue = queue.filter(q => q._rowIndex !== dataPayload._rowIndex);
                savePendingQueue(queue);
            } else {
                throw new Error("Respon server gagal");
            }
        } catch(err) {
            console.warn("Gagal terkirim langsung, masuk antrean offline:", err);
            queueForSync(dataPayload);
            showDynamicIsland('offline', '💾 Disimpan di HP (Sinyal Hilang)');
        }
    }

    function saveHistorySnapshot() {
        historyStack.push({
            alpha: [...tempAlpha],
            izin: [...tempIzin],
            sakit: [...tempSakit]
        });
        updateUndoUI();
    }

    function undoLastAction() {
        if (historyStack.length === 0) return;
        let lastState = historyStack.pop();
        tempAlpha = lastState.alpha;
        tempIzin = lastState.izin;
        tempSakit = lastState.sakit;
        renderChips();
        updateUndoUI();
        showToast("↩️ Perubahan dibatalkan!");
    }

    function updateUndoUI() {
        let btn = document.getElementById('btn-undo');
        if (!btn) return;
        if (historyStack.length > 0) {
            btn.disabled = false;
            btn.style.opacity = "1";
        } else {
            btn.disabled = true;
            btn.style.opacity = "0.5";
        }
    }

    function renderChips() {
        const setChips = (containerId, arr, type, chipClass) => {
            let el = document.getElementById(containerId);

        setChips('chips-izin', tempIzin, 'izin', 'chip-izin');
        setChips('chips-sakit', tempSakit, 'sakit', 'chip-sakit');

    }

    function moveChip(tgl, fromType, toType) {
        if (fromType === toType) return;
        saveHistorySnapshot();

        if (fromType === 'alpha') tempAlpha = tempAlpha.filter(x => x !== tgl);
        if (fromType === 'izin') tempIzin = tempIzin.filter(x => x !== tgl);
        if (fromType === 'sakit') tempSakit = tempSakit.filter(x => x !== tgl);

        if (toType === 'alpha' && !tempAlpha.includes(tgl)) tempAlpha.push(tgl);
        if (toType === 'izin' && !tempIzin.includes(tgl)) tempIzin.push(tgl);
        if (toType === 'sakit' && !tempSakit.includes(tgl)) tempSakit.push(tgl);

        renderChips();

        if (toType === 'trash') {
            showToast(`🗑️ Tgl ${tgl} dihapus. (Klik Undo jika batal)`);
        } else {
            showToast(`↔️ Tgl ${tgl} dipindah ke ${toType.toUpperCase()}`);
        }

        if (isPracticeMode) {
            showToast("🎉 Hebat! Kamu sudah bisa menggeser tanggal!");
            setTimeout(() => {
                closeEditModal();
                showToast("✅ Tutorial selesai! Selamat bertugas.");
                isPracticeMode = false;
            }, 1000);
        }
    }

    /* SISTEM DRAG & DROP TEROPTIMASI ULTRA-LANCAR */
    function setupTouchDragAndDrop() {
        let draggedData = null;
        let ghostEl = null;
        const trashZone = document.getElementById('trash-zone');
        let ticking = false;
        let currentX = 0, currentY = 0;
        let cachedZones = [];

        document.addEventListener('pointerdown', function(e) {
            let chip = e.target.closest('.chip');
            if (!chip) return;

            let tgl = parseInt(chip.getAttribute('data-tgl'), 10);
            let type = chip.getAttribute('data-type');
            if (!tgl || !type) return;

            draggedData = { tgl, type, element: chip };
            
            if (trashZone) trashZone.classList.add('show-floating');

            cachedZones = Array.from(document.querySelectorAll('[data-zone]')).map(el => {
                let rect = el.getBoundingClientRect();
                return {
                    element: el,
                    zone: el.getAttribute('data-zone'),
                    left: rect.left,
                    top: rect.top,
                    right: rect.right,
                    bottom: rect.bottom
                };
            });

            document.body.classList.add('is-dragging');

            ghostEl = chip.cloneNode(true);
            ghostEl.style.position = 'fixed';
            ghostEl.style.top = '0px';
            ghostEl.style.left = '0px';
            ghostEl.style.zIndex = '9999999';
            ghostEl.style.pointerEvents = 'none';
            ghostEl.style.opacity = '0.9';
            ghostEl.style.boxShadow = '0 10px 25px rgba(0,0,0,0.25)';
            
            currentX = e.clientX - 35;
            currentY = e.clientY - 20;
            ghostEl.style.transform = `translate3d(${currentX}px, ${currentY}px, 0) scale(1.1) rotate(2deg)`;
            
            document.body.appendChild(ghostEl);
            chip.style.opacity = '0.2';

            function updatePosition() {
                if (ghostEl) {
                    ghostEl.style.transform = `translate3d(${currentX}px, ${currentY}px, 0) scale(1.1) rotate(2deg)`;
                    
                    let checkX = currentX + 35;
                    let checkY = currentY + 20;
                    
                    cachedZones.forEach(z => {
                        let isHover = checkX >= z.left && checkX <= z.right && checkY >= z.top && checkY <= z.bottom;
                        if (isHover) {
                            z.element.classList.add('drag-over');
                        } else {
                            z.element.classList.remove('drag-over');
                        }
                    });
                }
                ticking = false;
            }

            function onPointerMove(pe) {
                if (!ghostEl) return;
                currentX = pe.clientX - 35;
                currentY = pe.clientY - 20;

                if (!ticking) {
                    requestAnimationFrame(updatePosition);
                    ticking = true;
                }
            }

            function onPointerUp(pe) {
                document.removeEventListener('pointermove', onPointerMove);
                document.removeEventListener('pointerup', onPointerUp);

                document.body.classList.remove('is-dragging');

                let dropZoneType = null;
                let checkX = pe.clientX;
                let checkY = pe.clientY;

                cachedZones.forEach(z => {
                    z.element.classList.remove('drag-over');
                    if (checkX >= z.left && checkX <= z.right && checkY >= z.top && checkY <= z.bottom) {
                        dropZoneType = z.zone;
                    }
                });

                if (trashZone) trashZone.classList.remove('show-floating');

                if (ghostEl) {
                    ghostEl.remove();
                    ghostEl = null;
                }
                if (draggedData && draggedData.element) {
                    draggedData.element.style.opacity = '1';
                }

                if (dropZoneType && draggedData) {
                    moveChip(draggedData.tgl, draggedData.type, dropZoneType);
                }
                draggedData = null;
            }

            document.addEventListener('pointermove', onPointerMove, { passive: true });
            document.addEventListener('pointerup', onPointerUp);
        });
    }

    function tambahTanggalGlobal(type) {
        const el = document.getElementById('input-global-tgl');
        if (!el) return;
        
        const rawVal = el.value.trim();
        if (!rawVal) return;

        const parsedDates = parseDates(rawVal);
        if (parsedDates.length === 0) {
            showToast("⚠️ Tanggal tidak valid (harus 1-31)");
            el.value = '';
            return;
        }

        saveHistorySnapshot();

        parsedDates.forEach(val => {
            tempAlpha = tempAlpha.filter(x => x !== val);
            tempIzin = tempIzin.filter(x => x !== val);
            tempSakit = tempSakit.filter(x => x !== val);

            if (type === 'alpha') tempAlpha.push(val);
            if (type === 'izin') tempIzin.push(val);
            if (type === 'sakit') tempSakit.push(val);
        });

        el.value = '';
        renderChips();
    }
